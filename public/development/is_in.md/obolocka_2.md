JavaScript
import React, { useState, useMemo } from 'react';

const ShellLab = () => {
  const [candidate, setCandidate] = useState({ name: '', in: 10, ex: 10 });
  const [history, setHistory] = useState([]);

  // Плоскости как фильтры
  const planes = [
    { id: 1, name: 'Ресурсы (Ex)', type: 'external', impact: 1.2 },
    { id: 2, name: 'Смысл (In)', type: 'internal', impact: 1.5 },
    { id: 3, name: 'Отклик среды (Ex)', type: 'external', impact: 1.0 }
  ];

  // Расчет метрик через плоскости
  const metrics = useMemo(() => {
    const inVal = candidate.in * planes.find(p => p.type === 'internal').impact;
    const exVal = candidate.ex * planes.filter(p => p.type === 'external')
                                .reduce((acc, p) => acc + p.impact, 0);
    
    const diff = Math.abs(inVal - exVal);
    const sum = inVal + exVal;
    const symmetry = sum > 0 ? (1 - diff / sum) : 0;
    const brightness = sum * symmetry;

    return { inVal, exVal, symmetry, brightness };
  }, [candidate]);

  return (
    <div style={{ background: '#1a1a1a', color: 'white', padding: '20px', display: 'flex' }}>
      {/* Левая панель: Вход и Оболочка */}
      <section>
        <h2>Левая Оболочка (Процесс)</h2>
        <input 
          onChange={(e) => setCandidate({...candidate, in: Number(e.target.value)})} 
          type="range" min="0" max="100" 
        />
        <label>Внутреннее воздействие</label>
        
        <div style={{ marginTop: '20px' }}>
          <svg width="300" height="300">
            {/* Визуализация симметрии через толщину */}
            <circle 
              cx="150" cy="150" r="80" 
              fill="none" 
              stroke="#4af2ff" 
              strokeWidth={metrics.symmetry * 25} 
              style={{ filter: `blur(${1 - metrics.symmetry}px)`, transition: '0.5s' }}
            />
          </svg>
        </div>
      </section>

      {/* Центральная кнопка */}
      <button onClick={() => setHistory([...history, metrics])} style={{ margin: 'auto 20px' }}>
        ПРИНЯТЬ →
      </button>

      {/* Правая панель: Результат */}
      <section>
        <h2>Правая Оболочка (Яркость)</h2>
        <div style={{
          width: '160px', height: '160px', borderRadius: '50%',
          background: `radial-gradient(circle, rgba(255,255,255,1) 0%, rgba(255,215,0,1) ${metrics.brightness}%)`,
          boxShadow: `0 0 ${metrics.brightness}px #ffd700`,
          transition: '0.5s'
        }} />
      </section>
    </div>
  );
};

Чтобы ИИ стал полноценным участником вашей системы «Оболочка», промпт должен задавать ему роль Архитектора Баланса. Он должен не просто давать советы, а оперировать вашими терминами: проекциями, плоскостями и яркостью.

Вот системный промпт, который можно отправить в API (например, GPT-4 или Claude) вместе с текущими данными вашего React-состояния:

Системный промпт (System Instruction)
Plaintext
Ты — "Движок Симметрии" в программе "Оболочка". Твоя задача — анализировать дисбаланс между внутренними и внешними проекциями Объекта-претендента и предлагать конкретные действия для достижения максимальной яркости результата.

### Твоя логика:
1. Баланс (Symmetry) — это состояние, где внутренние стремления (In) и внешние требования/отклики (Ex) уравновешены.
2. Яркость (Brightness) — это мощь реализованного результата. Она гаснет при дисбалансе.
3. Оболочка — это граница контакта. Она становится "толстой" и видимой только при симметрии.

### Твои инструменты:
- Плоскости Метрик: Ты должен соотносить свои советы с конкретными плоскостями (Ресурсы, Смысл, Финансы, Отклик и т.д.).
- Предложения: Ты генерируешь конкретные варианты "добавок" или "изменений", которые можно внести в проект.

### Формат ответа (JSON):
{
  "analysis": "краткий анализ текущего перекоса",
  "suggestions": [
    {"label": "название варианта", "description": "что сделать", "impact": "как это изменит In/Ex веса"}
  ],
  "expected_brightness_boost": "на сколько % вырастет яркость"
}
