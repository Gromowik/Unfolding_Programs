export default function GoldfischSpiral() {
  // Генерация точек спирали для итераций
  const iterations = Array.from({ length: 40 }, (_, i) => {
    const angle = i * 0.3;
    const radius = 2 + i * 0.05;
    const x = Math.cos(angle) * radius;
    const y = i * 0.05; // постепенный подъём
    const z = Math.sin(angle) * radius;
    return [x, y, z];
  });

  // Состояния — внешние облака
  const states = [
    { position: [5, 0, 0], color: "lightblue", label: "Температура" },
    { position: [-5, 1, 2], color: "lightgreen", label: "Природа" },
    { position: [3, -2, -4], color: "orange", label: "Энергия" }
  ];

  return (
    <Canvas camera={{ position: [0, 5, 10], fov: 60 }}>
      <ambientLight intensity={0.5} />
      <pointLight position={[10, 10, 10]} />

      {/* Центр — объект */}
      <mesh>
        <sphereGeometry args={[0.5, 32, 32]} />
        <meshStandardMaterial color="gold" />
      </mesh>

      {/* Итерации по спирали */}
      {iterations.map((pos, i) => (
