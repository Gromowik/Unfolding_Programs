        const VESSEL_KEY = 'AXIOM_VESSEL';
        let iterationCounter = 1;

        // --- 1. ФУНКЦИИ ХРАНЕНИЯ (Сосуд Опыта) ---

        function loadVessel() {
            const data = localStorage.getItem(VESSEL_KEY);
            return data ? JSON.parse(data) : [];
        }

        function saveProjection(projection) {
            const vessel = loadVessel();
            vessel.push(projection);
            localStorage.setItem(VESSEL_KEY, JSON.stringify(vessel));
        }

        // --- 2. ФУНКЦИИ ОТОБРАЖЕНИЯ (Облако Сходств) ---

        function renderSimilarityCloud() {
            const vessel = loadVessel();
            const list = document.getElementById('cloud-list');
            list.innerHTML = ''; // Очищаем список перед обновлением

            if (vessel.length === 0) {
                list.innerHTML = '<li style="border: none;">Ваш Сосуд Опыта пока пуст. Сгенерируйте первую Проекцию!</li>';
                return;
            }

            // Рендерим все сохраненные Герои
            vessel.forEach(proj => {
                const heroName = proj.M.O.name;
                
                // Извлекаем методы из L-связей
                const plusMethod = proj.L.find(l => l.type[1] === ' + ').type[3];
                const minusMethod = proj.L.find(l => l.type[1] === ' - ').type[3];
                
                const listItem = document.createElement('li');
                listItem.innerHTML = `
                    <strong>${heroName}</strong><br>
                    M+ (Симбиоз): ${plusMethod.split(',')[0].trim()}<br>
                    M- (Итерация): ${minusMethod.split(',')[0].trim()}
                `;
                list.appendChild(listItem);
            });
        }
        
        // --- 3. ОСНОВНАЯ ЛОГИКА ГЕНЕРАЦИИ ---

        function generateId(prefix) {
            const now = new Date();
            const timePart = `${now.getHours()}:${now.getMinutes()}:${now.getSeconds()}_${iterationCounter++}`;
            return `2025-11-01T${timePart}_${prefix}`;
        }

        function updateImportance(type) {
            const slider = document.getElementById(`${type}-importance`);
            const valueDisplay = document.getElementById(`${type}-importance-value`);
            valueDisplay.textContent = slider.value;
        }

        function generateProjection() {
            const concept = document.getElementById('concept-input').value.toUpperCase() || 'НОВЫЙ_ГЕРОЙ';
            const plusMethod = document.getElementById('plus-method').value || 'МЕТОД_ПЛЮСА';
            const minusMethod = document.getElementById('minus-method').value || 'МЕТОД_МИНУСА';
            const plusText = document.getElementById('plus-input').value || 'Фокус Плюса (Содержание)';
            const minusText = document.getElementById('minus-input').value || 'Фокус Минуса (Процесс)';
            
            const plusImportance = parseInt(document.getElementById('plus-importance').value);
            const minusImportance = parseInt(document.getElementById('minus-importance').value);

            // Генерация ID для графа
            const heroId = generateId(`HERO_${concept}`);
            const plusMethodId = generateId('METHOD_PLUS');
            const minusMethodId = generateId('METHOD_MINUS');
            const plusFocusId = generateId('FOCUS_PLUS');
            const minusFocusId = generateId('FOCUS_MINUS');

            // Обновление HTML (Визуальный результат)
            document.getElementById('concept-display').textContent = `Текущий Герой: ${concept}`;
            document.getElementById('plus-method-out').textContent = plusMethod;
            document.getElementById('minus-method-out').textContent = minusMethod;
            document.getElementById('plus-text').textContent = plusText;
            document.getElementById('minus-text').textContent = minusText;
            document.getElementById('plus-title-out').textContent = plusMethod.split(',')[0].trim();
            document.getElementById('minus-title-out').textContent = minusMethod.split(',')[0].trim();


            // Генерация Финальной JSON-Структуры
            const jsonStructure = {
                "M": {
                    "id": heroId,
                    "O": {
                        "text": `Бинарный анализ Героя ${concept} через Методы: ${plusMethod} против ${minusMethod}.`,
                        "name": concept
                    },
                    "type": ["Hero", {"inclusions": [concept, plusMethod, minusMethod]}],
                    "Parent": null 
                },
                "L": [
                    {
                        "type": ["M+", " + ", plusMethodId, plusMethod, "Human"], 
                        "importance": plusImportance 
                    },
                    {
                        "type": ["M+", " - ", minusMethodId, minusMethod, "Human"],
                        "importance": minusImportance 
                    },
                    {
                        "type": ["m", " + ", plusFocusId, plusMethodId, "Human"], 
                        "importance": plusImportance
                    },
                    {
                        "type": ["m", " - ", minusFocusId, minusMethodId, "Human"],
                        "importance": minusImportance
                    }
                ]
            };
            
            // 4. СОХРАНЕНИЕ И ОБНОВЛЕНИЕ
            saveProjection(jsonStructure);
            renderSimilarityCloud();
            
            // Отображение JSON
            document.getElementById('json-output').textContent = JSON.stringify(jsonStructure, null, 4);
        }
