🔧 HTML + CSS
html
<!DOCTYPE html>
<html lang="ru">
<head>
  <meta charset="UTF-8">
  <title>Размещение прямоугольников</title>
  <style>
    #container {
      position: relative;
      border: 2px dashed #999;
      margin: 20px auto;
      background: #f5f5f5;
    }
    .box {
      position: absolute;
      background: rgba(100, 150, 240, 0.7);
      border: 1px solid #444;
      box-sizing: border-box;
    }
  </style>
</head>
<body>
  <div id="container"></div>

  <script>
    // 👉 Размеры прямоугольников [ширина, высота]
    const boxes = [
      [40, 60], [80, 90], [30, 30],
      [50, 70], [60, 50], [100, 40],
      [25, 25], [80, 80], [45, 55]
    ];

    // 📐 Вычисляем общую площадь
    const totalArea = boxes.reduce((sum, [w, h]) => sum + w * h, 0);

    // 🔲 Предполагаемый размер квадрата
    const side = Math.ceil(Math.sqrt(totalArea));
    const container = document.getElementById('container');
    container.style.width = side + 'px';
    container.style.height = side + 'px';

    // ➿ Размещение элементов построчно
    let x = 0, y = 0, rowHeight = 0;

    boxes.forEach(([w, h]) => {
      if (x + w > side) {
        // переход на новую строку
        x = 0;
        y += rowHeight;
        rowHeight = 0;
      }

      const div = document.createElement('div');
      div.className = 'box';
      div.style.width = w + 'px';
      div.style.height = h + 'px';
      div.style.left = x + 'px';
      div.style.top = y + 'px';

      container.appendChild(div);

      x += w;
      rowHeight = Math.max(rowHeight, h);
    });
  </script>
</body>
</html>

🧠 JavaScript: подбор лучшего расклада по форме
html
<!DOCTYPE html>
<html lang="ru">
<head>
  <meta charset="UTF-8">
  <title>Адаптивная раскладка</title>
  <style>
    #bestLayout {
      position: relative;
      border: 2px dashed #333;
      margin: 20px auto;
      background-color: #f0f0f0;
    }
    .block {
      position: absolute;
      background: rgba(200, 100, 240, 0.7);
      border: 1px solid #333;
      box-sizing: border-box;
    }
  </style>
</head>
<body>
  <h2>Лучшая приближенность к квадрату</h2>
  <div id="bestLayout"></div>

  <script>
    // Размеры блоков
    const containers = [
      [80, 40], [60, 60], [100, 50],
      [50, 70], [90, 30], [40, 40],
      [70, 60], [30, 30], [100, 20], [60, 90]
    ];

    const totalArea = containers.reduce((sum, [w, h]) => sum + w * h, 0);
    const baseSide = Math.ceil(Math.sqrt(totalArea));
    let bestRatioDiff = Infinity;
    let bestLayout = [];

    for (let groupSize = 1; groupSize <= containers.length; groupSize++) {
      let x = 0, y = 0, rowHeight = 0;
      let layout = [];
      let maxX = 0, maxY = 0;

      containers.forEach(([w, h], i) => {
        if (i % groupSize === 0 && i !== 0) {
          // новая строка
          x = 0;
          y += rowHeight;
          rowHeight = 0;
        }

        layout.push({ left: x, top: y, width: w, height: h });
        x += w;
        rowHeight = Math.max(rowHeight, h);
        maxX = Math.max(maxX, x);
        maxY = Math.max(maxY, y + h);
      });

      const ratio = maxX / maxY;
      const diff = Math.abs(ratio - 1); // насколько далеко от квадрата

      if (diff < bestRatioDiff) {
        bestRatioDiff = diff;
        bestLayout = layout;
      }
    }

    // Отрисовка лучшего варианта
    const container = document.getElementById('bestLayout');
    container.style.width = baseSide + 'px';
    container.style.height = baseSide + 'px';

    bestLayout.forEach(box => {
      const div = document.createElement('div');
      div.className = 'block';
      div.style.left = box.left + 'px';
      div.style.top = box.top + 'px';
      div.style.width = box.width + 'px';
      div.style.height = box.height + 'px';
      container.appendChild(div);
    });
  </script>
</body>
</html>
